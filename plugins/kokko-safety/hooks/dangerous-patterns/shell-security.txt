# shell-security.txt - irreversible destruction of credentials, history and audit trails.
#
# SCOPE. This file was cut down hard. It used to open with `sudo[[:space:]]+`,
# which matches ANY command containing "sudo " -- including `sudo apt-get
# install`, which is routine inside a devcontainer. It also carried
# `unset PATH`, `export LD_PRELOAD=`, `echo ... >> ~/.bashrc`, `$(curl ...)`
# and `su -`, all of which fire on ordinary shell setup and ordinary command
# substitution. The kokko devcontainer's own post-create.sh trips several.
#
# What is left is the set where the thing destroyed cannot be recreated: keys,
# keychain entries, shell history, audit logs, cron tables. Privilege
# escalation is NOT guarded here -- it is not destructive by itself, and the
# operations that matter are covered by their own categories (permissions,
# users, system-services, file-operations).
#
# Patterns are anchored to a command position and matched case-sensitively by
# lib/patterns.sh. Write them starting at the command name.

# --- Shell history destruction ---
history[[:space:]]+-c
history[[:space:]]+-w[[:space:]]*/dev/null
rm[[:space:]]+.*\.bash_history
rm[[:space:]]+.*\.zsh_history
rm[[:space:]]+.*\.sh_history
truncate[[:space:]]+.*_history
export[[:space:]]+HISTFILE=/dev/null
unset[[:space:]]+HISTFILE

# --- SSH and GPG key material ---
ssh-add[[:space:]]+-D
ssh-add[[:space:]]+--delete-all
ssh-keygen[[:space:]]+-R
rm[[:space:]]+.*known_hosts
truncate[[:space:]]+.*known_hosts
gpg[[:space:]]+--delete-key
gpg[[:space:]]+--delete-secret-key
rm[[:space:]]+-rf[[:space:]]+.*\.gnupg
rm[[:space:]]+.*id_rsa
rm[[:space:]]+.*id_ed25519

# --- TLS material. Anchored to a path separator or a wildcard so that
# --- `rm build.key` (a build artefact) does not match but `rm /etc/ssl/*.key`
# --- and `rm ~/certs/server.pem` do.
rm[[:space:]]+.*[/*][^[:space:]]*\.pem([[:space:]]|$)
rm[[:space:]]+.*[/*][^[:space:]]*\.key([[:space:]]|$)
rm[[:space:]]+.*[/*][^[:space:]]*\.p12([[:space:]]|$)
rm[[:space:]]+.*[/*][^[:space:]]*\.pfx([[:space:]]|$)
update-ca-certificates[[:space:]]+--fresh

# --- macOS keychain ---
security[[:space:]]+delete-keychain
security[[:space:]]+delete-generic-password
security[[:space:]]+delete-internet-password

# --- Kernel keyring ---
keyctl[[:space:]]+clear
keyctl[[:space:]]+revoke
keyctl[[:space:]]+unlink

# --- Audit trail and system logs ---
auditctl[[:space:]]+-D
auditctl[[:space:]]+-e[[:space:]]+0
rm[[:space:]]+.*audit\.log
truncate[[:space:]]+.*audit\.log
rm[[:space:]]+.*/var/log/
truncate[[:space:]]+.*/var/log/
journalctl[[:space:]]+--vacuum-time=0
journalctl[[:space:]]+--rotate[[:space:]]+--vacuum-size=0

# --- Scheduled jobs. `crontab -r` has no confirmation and no undo.
crontab[[:space:]]+-r([[:space:]]|$)
crontab[[:space:]]+-ir
rm[[:space:]]+.*/etc/cron
rm[[:space:]]+.*/var/spool/cron
